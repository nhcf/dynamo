976        run_update_callbacks(&registries);

981        // has no unit field, so it would otherwise be dropped here.
982        let mut units: HashMap<String, String> = HashMap::new();
983        let mut prom_types: HashMap<String, String> = HashMap::new();
984        for registry in &registries {
985            for built in registry.execute_typed_callbacks() {
986                let name = built.family.name().to_string();
987                // Publish the side metadata only once the family is accepted.
988                // The maps are keyed by name, so metadata from a family the
989                // merger rejects would otherwise be applied to the family that
990                // did survive under that name -- a rejected `info` could make a
991                // surviving gauge encode as a non-monotonic Sum.
992                if let Err(error) = merger.add_family(built.family) {
993                    tracing::warn!(metric_name = %name, %error, "skipping typed metric family");
994                    continue;
995                }
996                // First writer wins, matching how the merger itself resolves a
997                // conflict: the first family fixes help and type, and later
998                // ones must agree. A later family can be accepted and still
999                // have every sample dropped as a duplicate series, so letting
1000                // it overwrite the unit would label the surviving data with a
1001                // unit from a source that contributed none of it.

1003                    units.entry(name.clone()).or_insert(built.unit);
1004                }
1005                prom_types.entry(name).or_insert(built.prom_type);
1006            }
1007        }

565/// closure, so a cancelled request cannot release the gate before its render

为什么 spawn_blocking？

采集回调可能等待 Python GIL，不能让同步工作阻塞 Tokio 异步执行线程。OTLP 路径等待 collection 完成再处理关闭，因为已启动的 blocking task 无法直接取消；否则 Python teardown 后可能仍有任务访问 GIL。

当前 Runtime 中，配置解析错误进入 Err 分支，记录 misconfigured; not exporting，然后继续初始化。不能把“协议只支持 gRPC、解析返回错误”推断为“整个进程必然启动失败”。
