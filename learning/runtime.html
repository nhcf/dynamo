285                endpoint = %endpoint_name_for_cleanup,
286                "Unregistering endpoint from request plane server"
287            );
288
289            if let Err(e) = server_for_cleanup
290                .unregister_endpoint_instance(&endpoint_id, connection_id)
291                .await
292            {
293                tracing::warn!(
294                    endpoint = %endpoint_name_for_cleanup,
295                    error = %e,
296                    "Failed to unregister endpoint"
297                );
298            }
299
300            if let Some(tracker) = tracker_clone {
301                tracing::debug!("Unregister endpoint from graceful shutdown tracker");
302                tracker.unregister_endpoint();
303            }
304
305            anyhow::Ok(())
306        });
307
308        Ok(StartedEndpoint {
309            instance,
310            shutdown_token: endpoint_shutdown_token,
311            task,
312        })
313    }

关闭先阻止新流量，再等待已在途请求与后台清理。撤销发现有传播延迟；对端旧连接仍可能在途。Stop、Task Drop、endpoint unregister 与 engine.cleanup 有不同 owner，SIGKILL 不会执行这些正常路径。完整 drain/cleanup 预算见可靠性专题。
