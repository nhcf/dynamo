126    /// Linear prefix match: walks `seq_hash` left-to-right, stopping on
127    /// the first hash that hits neither the active nor the inactive pool.
128    ///
129    /// The whole active-or-inactive prefix is resolved under a **single**
130    /// store-mutex acquisition via `BlockStore::match_prefix_locked_batch`
131    /// — no per-hash registry radix-tree lookup, no per-hash store lock.
132    /// Frequency-tracker touches are batched and applied *after* the store
133    /// lock is released: every returned block is touched exactly once
134    /// (including inactive resurrections).
135    pub fn match_blocks(&self, seq_hash: &[SequenceHash]) -> Vec<ImmutableBlock<T>> {
136        self.metrics
137            .inc_match_hashes_requested(seq_hash.len() as u64);
138
139        if seq_hash.is_empty() {
140            self.metrics.inc_match_blocks_returned(0);
141            return Vec::new();
142        }
143
144        // ONE store-lock acquisition for the whole active+inactive prefix.
145        let inners = self.store.match_prefix_locked_batch(seq_hash);
146
147        // Frequency-tracker touches, batched, AFTER the store lock is
148        // released. Touches every returned hit exactly once — including
149        // inactive resurrections, which the old `find_inactive_primaries`
150        // path never touched.
151        if self.block_registry.has_frequency_tracking() {
152            for inner in &inners {
153                self.block_registry.touch(inner.sequence_hash());
154            }
155        }
156
157        let matched: Vec<ImmutableBlock<T>> =
158            inners.into_iter().map(ImmutableBlock::from_inner).collect();
159
160        self.metrics.inc_match_blocks_returned(matched.len() as u64);
161        tracing::debug!(
162            num_hashes = seq_hash.len(),
163            total_matched = matched.len(),
164            "match_blocks result"
165        );
166        tracing::trace!(matched = ?matched, "matched blocks");
167        matched
168    }

match_blocks 沿 sequence hash 左到右查 active/inactive，遇到 miss 停止；scan/match_scattered 是不同接口。返回命中句柄同时保护实际块，后续 Drop 再释放。TinyLFU 或 LRU 等策略影响淘汰，但不改变 hash 域兼容条件；性能结论需要具体池策略和 workload。

333    def commit(self) -> bool:
334        """Synchronize, unmap writer mappings, then commit.
335
336        Commit is a publish barrier. It guarantees all prior GPU writes in the
337        current context are complete before the server transitions state. After
338        a successful commit, the former writer process no longer has any mapped
339        access to the published allocations. Any failure after local unmap
340        raises because the process cannot safely recover its CUDA VMM state.
341        """
342        self._require_rw()
343
344        # Publish barrier: all writer-side GPU work must be visible before commit.
345        self._vmm.synchronize()
346
347        for mapping in list(self._mappings.values()):
348            if mapping.handle != 0:
349                self.unmap_va(mapping.va)
350
351        self._va_preserved = True
352        self._unmapped = True
353
354        self._client_rpc.commit()
355        self._client = None
356        self._granted_lock_type = None
357        return True

359    def commit_layout(self) -> "LayoutCommit":
360        """Seal the allocation set: the shape is final, the pages outlive this session.
361
362        The counterpart to :meth:`commit`, which publishes *contents*: it unmaps the
363        writer, closes the session, and lets readers attach. This publishes only the
364        *shape*, leaving mappings and session intact so the caller keeps writing.
365
366        Call it once the pool is built. That call is the atomic boundary: die before it
367        and the half-built pool is discarded, die after and a standby can adopt it.
368
369        The session is narrowed to RW_DATA, so allocate and free now raise. Reconnect
370        with RW to build a different layout.
371        """
372        self._require_rw()
373        # Publish barrier, matching commit(): make this process's GPU writes visible
374        # before the layout is advertised as reattachable.
375        self._vmm.synchronize()
376        response = self._client.commit_layout()
377        # The server decides what we hold now; read it back rather than assuming.
378        self._granted_lock_type = self._client.lock_type
379        self._last_memory_layout_hash = response.memory_layout_hash
380        return LayoutCommit(
381            memory_layout_hash=response.memory_layout_hash,
382            granted_lock_type=self._granted_lock_type,
383        )

commit 先同步 GPU 写、撤销 writer 映射再发布内容并结束 session；commit_layout 只发布可重接的布局，保留写入映射并收窄为 RW_DATA，阻止后续 allocate/free 改变结构。这两种原子边界不同，不能把 layout committed 当 tensor 内容永远不会再变，也不能绕过客户端同步直接发布。
