75    async def _forward_with_priority_retry(
76        self,
77        request: Dict[str, Any],
78        request_type: str,
79        initial_pool_idx: int,
80        namespaces: List[str],
81        clients: Dict[str, Client],
82        pool_priorities: List[int],
83        context=None,
84    ) -> AsyncGenerator[Dict[str, Any], None]:
85        """
86        Forward a request to the selected pool and retry faster pools on failure.
87
88        Retry order is controlled by pool priorities: lower priority numbers are
89        faster, so retries walk from the selected pool toward faster pools.
90        """
91        pool_order = get_priority_retry_order(
92            selected_pool=initial_pool_idx,
93            pool_priorities=pool_priorities,
94            enable_priority_retry=self.config.enable_priority_retry,
95        )
96        # The egress codec is chosen per destination; a JSON destination would read
97        # a packed buffer as one id per byte, so forward a list.
98        request = normalize_request_token_ids(dict(request))
99
100        for attempt_idx, pool_idx in enumerate(pool_order):
101            namespace = namespaces[pool_idx]
102            client = clients[namespace]
103            yielded_output = False
104
105            try:
106                stream = await client.generate(request, context=context)
107                async for output in stream:
108                    yielded_output = True
109                    data = output.data() if hasattr(output, "data") else output
110                    yield data
111                return
112            except Exception as e:
113                if context is not None and context.is_stopped():
114                    logger.debug(
115                        "Request context was cancelled; skipping priority retry"
116                    )
117                    raise
118
119                is_last_attempt = attempt_idx == len(pool_order) - 1
120                if yielded_output:
121                    logger.error(
122                        f"Error forwarding {request_type} request to {namespace} "
123                        f"after streaming started; cannot safely retry: {e}"
124                    )
125                    raise
126
127                if is_last_attempt:
128                    logger.error(
129                        f"Error forwarding {request_type} request to {namespace}; "
130                        f"no priority retry pools remain: {e}"
131                    )
132                    raise
133
134                next_pool_idx = pool_order[attempt_idx + 1]
135                next_namespace = namespaces[next_pool_idx]
136                if request_type == "decode":
137                    # A failed decode attempt may already have caused the prefill
138                    # engine to retire or drop its KV cache for this request. That
139                    # is acceptable: the retry is a fresh decode request to another
140                    # pool. The backend should handle any cache miss normally, but
141                    # current Dynamo backends do not support this yet.
142                    logger.debug("Retrying decode request after pool failure")
143                logger.warning(
144                    f"Error forwarding {request_type} request to pool {pool_idx} "
145                    f"({namespace}): {e}; retrying faster pool {next_pool_idx} "
146                    f"({next_namespace})"
147                )

priority 数字越小表示更快 pool；可按配置向更快候选重试。已有任何输出后不能安全重试，取消也不继续迁移。源码明确提示 Decode retry 可能遇到 Prefill KV 已退休，而当前 backend 尚不支持该缺失恢复语义；不能把存在 retry 代码当作 PD 跨池无损故障转移保证。

官方 Global Router README 列出 vLLM、Mocker、SGLang 路径；TensorRT-LLM 没有专用端到端集成验证。共享 Frontend 能处理 TRT PD，不自动代表该全局拓扑受支持。
